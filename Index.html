  <script>

    const people = [

      { name: "Alvin Thomas", key: "alvin" },

      { name: "Mayank Singh", key: "mayank" },

      { name: "Oommen John", key: "oommen" }

    ];


    const STORAGE_KEY = "oerliveMonthlyArticleRepositoryV1";


    const defaultData = {

      months: {}

    };


    let repository = JSON.parse(localStorage.getItem(STORAGE_KEY)) || defaultData;


    function getCurrentMonthKey() {

      const now = new Date();

      const year = now.getFullYear();

      const month = String(now.getMonth() + 1).padStart(2, "0");

      return `${year}-${month}`;

    }


    function getMonthLabel(monthKey) {

      const [year, month] = monthKey.split("-");

      const date = new Date(Number(year), Number(month) - 1, 1);


      return date.toLocaleDateString("en-GB", {

        month: "long",

        year: "numeric"

      });

    }


    function createEmptyMonth() {

      return {

        alvin: [],

        mayank: [],

        oommen: []

      };

    }


    function ensureCurrentMonthExists() {

      const currentMonth = getCurrentMonthKey();


      if (!repository.months[currentMonth]) {

        repository.months[currentMonth] = createEmptyMonth();

        saveRepository();

      }

    }


    function getCurrentMonthData() {

      ensureCurrentMonthExists();

      return repository.months[getCurrentMonthKey()];

    }


    function saveRepository() {

      localStorage.setItem(STORAGE_KEY, JSON.stringify(repository));

    }


    function isValidUrl(value) {

      try {

        const url = new URL(value);

        return url.protocol === "http:" || url.protocol === "https:";

      } catch {

        return false;

      }

    }


    function normalizeUrl(value) {

      return value.trim();

    }


    function getAllLinksAcrossRepository() {

      const allLinks = [];


      Object.values(repository.months).forEach(monthData => {

        people.forEach(person => {

          monthData[person.key].forEach(link => allLinks.push(link.toLowerCase()));

        });

      });


      return allLinks;

    }


    function showToast(message) {

      const toast = document.getElementById("toast");

      toast.textContent = message;

      toast.style.display = "block";


      setTimeout(() => {

        toast.style.display = "none";

      }, 2800);

    }


    function getLinkTitle(url) {

      try {

        const parsed = new URL(url);

        let path = parsed.pathname

          .replaceAll("/", " ")

          .replaceAll("-", " ")

          .replaceAll("_", " ")

          .trim();


        if (!path) {

          return parsed.hostname;

        }


        return path.length > 72 ? path.substring(0, 72) + "..." : path;

      } catch {

        return url;

      }

    }


    function getTotalForMonth(monthData) {

      return people.reduce((sum, person) => {

        return sum + monthData[person.key].length;

      }, 0);

    }


    function renderBarChart(containerId, monthData) {

      const container = document.getElementById(containerId);

      const maxCount = Math.max(...people.map(person => monthData[person.key].length), 1);


      container.innerHTML = people.map(person => {

        const count = monthData[person.key].length;

        const width = (count / maxCount) * 100;


        return `

          <div class="bar-row">

            <div class="bar-name">${person.name}</div>

            <div class="bar-track">

              <div class="bar-fill" style="width: ${width}%"></div>

            </div>

            <div class="bar-value">${count}</div>

          </div>

        `;

      }).join("");

    }


    function renderCards() {

      const container = document.getElementById("cardsContainer");

      const currentMonthData = getCurrentMonthData();


      container.innerHTML = "";


      const counts = people.map(person => currentMonthData[person.key].length);

      const maxCount = Math.max(...counts, 1);


      people.forEach(person => {

        const links = currentMonthData[person.key];

        const count = links.length;

        const percentage = (count / maxCount) * 100;


        const listHtml = links.length

          ? links.map((link, index) => `

              <div class="story-item">

                <a href="${link}" target="_blank" title="${link}">

                  ${index + 1}. ${getLinkTitle(link)}

                </a>

                <button class="delete-btn" onclick="removeStory('${person.key}', ${index})">Remove</button>

              </div>

            `).join("")

          : `<div class="empty-state">No story links added for this month yet.</div>`;


        const card = document.createElement("div");

        card.className = "person-card";


        card.innerHTML = `

          <div class="person-name">${person.name}</div>


          <div class="article-count">${count}</div>

          <div class="article-label">${count === 1 ? "Story Link This Month" : "Story Links This Month"}</div>


          <div class="mini-bar">

            <div class="mini-bar-fill" style="width: ${percentage}%"></div>

          </div>


          <div class="link-form">

            <input

              type="url"

              id="input-${person.key}"

              placeholder="Paste OERLive story link here"

              onkeydown="handleEnter(event, '${person.key}')"

            />

            <button class="submit-btn" onclick="addStory('${person.key}')">Add Link</button>

          </div>


          <div class="links-section">

            <div class="links-title">Stories posted by ${person.name}</div>

            <div class="story-list">

              ${listHtml}

            </div>

          </div>

        `;


        container.appendChild(card);

      });

    }


    function addStory(key) {

      const input = document.getElementById(`input-${key}`);

      const link = normalizeUrl(input.value);


      if (!link) {

        showToast("Please paste a story link first.");

        return;

      }


      if (!isValidUrl(link)) {

        showToast("Please enter a valid URL beginning with http:// or https://");

        return;

      }


      const allLinks = getAllLinksAcrossRepository();


      if (allLinks.includes(link.toLowerCase())) {

        showToast("This story link has already been added.");

        return;

      }


      const currentMonthData = getCurrentMonthData();

      currentMonthData[key].push(link);


      input.value = "";

      saveRepository();

      refreshDashboard();


      showToast("Story link added. Monthly ticker updated.");

    }


    function removeStory(key, index) {

      const confirmRemove = confirm("Remove this story link from this month's count?");

      if (!confirmRemove) return;


      const currentMonthData = getCurrentMonthData();

      currentMonthData[key].splice(index, 1);


      saveRepository();

      refreshDashboard();


      showToast("Story link removed. Monthly count updated.");

    }


    function handleEnter(event, key) {

      if (event.key === "Enter") {

        addStory(key);

      }

    }


    function updateSummary() {

      const currentMonth = getCurrentMonthKey();

      const currentMonthData = getCurrentMonthData();

      const total = getTotalForMonth(currentMonthData);


      document.getElementById("totalArticles").textContent = total;

      document.getElementById("currentMonthBadge").textContent = getMonthLabel(currentMonth);

      document.getElementById("monthSummary").textContent =

        `${getMonthLabel(currentMonth)} publishing count, calculated only from submitted story links.`;


      let leader = people[0];


      people.forEach(person => {

        if (currentMonthData[person.key].length > currentMonthData[leader.key].length) {

          leader = person;

        }

      });


      document.getElementById("leaderName").textContent = total === 0 ? "-" : leader.name;


      document.getElementById("leaderScore").textContent =

        total === 0

          ? "0 stories"

          : `${currentMonthData[leader.key].length} ${currentMonthData[leader.key].length === 1 ? "story" : "stories"}`;


      updateTicker(total, currentMonthData);

      renderBarChart("currentMonthChart", currentMonthData);

    }

    function renderHistoryRepository() {

      const monthList = document.getElementById("monthList");

      const historySelect = document.getElementById("historySelect");

      const currentMonth = getCurrentMonthKey();


      const monthKeys = Object.keys(repository.months)

        .sort()

        .reverse();


      historySelect.innerHTML = `<option value="">Select a month</option>`;


      monthKeys.forEach(monthKey => {

        const option = document.createElement("option");

        option.value = monthKey;

        option.textContent = monthKey === currentMonth

          ? `${getMonthLabel(monthKey)} - Current`

          : getMonthLabel(monthKey);


        historySelect.appendChild(option);

      });


      const archivedMonths = monthKeys.filter(monthKey => monthKey !== currentMonth);


      if (!archivedMonths.length) {

        monthList.innerHTML = `

          <div class="empty-state">

            No previous month archives yet. Once a new month begins, the earlier month will remain visible here.

          </div>

        `;

        return;

      }


      monthList.innerHTML = archivedMonths.map(monthKey => {

        const monthData = repository.months[monthKey];

        const total = getTotalForMonth(monthData);


        return `

          <div class="month-item" onclick="selectHistoryMonth('${monthKey}')" id="month-${monthKey}">

            <strong>${getMonthLabel(monthKey)}</strong>

            <small>${total} verified stories archived</small>

          </div>

        `;

      }).join("");

    }


    function selectHistoryMonth(monthKey) {

      document.getElementById("historySelect").value = monthKey;

      renderSelectedHistory();

    }


    function renderSelectedHistory() {

      const selectedMonth = document.getElementById("historySelect").value;

      const historyTitle = document.getElementById("historyTitle");

      const historyChart = document.getElementById("historyChart");

      const historyNote = document.getElementById("historyNote");


      document.querySelectorAll(".month-item").forEach(item => {

        item.classList.remove("active");

      });


      if (!selectedMonth || !repository.months[selectedMonth]) {

        historyTitle.textContent = "Select a month to view archive";

        historyChart.innerHTML = "";

        historyNote.textContent =

          "When a new month begins, the previous month remains saved here automatically.";

        return;

      }


      const selectedItem = document.getElementById(`month-${selectedMonth}`);

      if (selectedItem) selectedItem.classList.add("active");


      const monthData = repository.months[selectedMonth];

      const total = getTotalForMonth(monthData);


      historyTitle.textContent = `${getMonthLabel(selectedMonth)} Performance`;

      renderBarChart("historyChart", monthData);


      historyNote.innerHTML = `

        Total stories in ${getMonthLabel(selectedMonth)}: <strong>${total}</strong><br>

        Alvin Thomas: <strong>${monthData.alvin.length}</strong>,

        Mayank Singh: <strong>${monthData.mayank.length}</strong>,

        Oommen John: <strong>${monthData.oommen.length}</strong>.

      `;

    }


    function refreshDashboard() {

      ensureCurrentMonthExists();

      renderCards();

      updateSummary();

      renderHistoryRepository();

      renderSelectedHistory();

    }

        minute: "2-digit",

        second: "2-digit"

      });

    }
